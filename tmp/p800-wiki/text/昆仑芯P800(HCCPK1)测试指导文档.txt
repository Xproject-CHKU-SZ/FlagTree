3.2 挂载本地NVME盘（可选）
如果系统盘较小，比如50GB以下，建议挂载本地NVME盘，并将容器镜像的默认保存路径修改到本地NVME盘（所有训练节点都需要更新）。
# 挂载nvme1n1
printf "n\np\n1\n\n\nw\n" | fdisk /dev/nvme1n1
sleep 1
mkfs.ext4 /dev/nvme1n1p1
#如果磁盘分区已经存在，可以直接挂载
mkdir -p /data1 && mount /dev/nvme1n1p1 /data1
df -h

#安装docker
sudo yum install -y docker-ce docker-ce-cli containerd.io --allowerasing

# （可选）修改fstab，添加系统重启之后自动挂载操作
echo "/dev/nvme1n1p1 /data1 ext4 defaults 0 0" >> /etc/fstab

# 修改docker默认保存路径
mv /var/lib/docker /data1/ && ln -sf /data1/docker /var/lib/docker
systemctl restart docker

带宽测试
服务器总共包含8张网卡，下面以mlx5_bond_0为例来展示：
带宽测试server：
numactl --cpunodebind=0 --membind=0 ib_write_bw -d mlx5_bond_0 -x 3 -F --report_gbits -p 18500  -q 16 --run_infinitely


带宽测试client：
# 注意：29.74.124.134为server端的mlx5_bond_0的IP
# 可以通过在server端执行「ifconfig bond0」来获取
numactl --cpunodebind=0 --membind=0 ib_write_bw -d mlx5_bond_0 -x 3 -F --report_gbits -p 18500  -q 16 --run_infinitely 29.74.124.134


预期输出：


注意：
其中client侧需要指定server端IP, cpunodebind，membind绑核需要与网卡numa node一致, 并且由于pcie switch限制，若同一pcie switch上插了多张卡（比如bond0和bond1的网卡插在同一pcie switch上，则bond0和bond1需要错开测试），网卡需要错开pcie switch验证, 例如bond0/2/4/6同时测，bond1/3/5/7同时测；若每个bond的网卡有独立的pcie switch，可以同时并发测8个bond，达到3.2T带宽（pcie switch分布可以通过lspci -tv命令查询）
由于单个QP只能使用一个端口，所以需要配置多个QP端口将bond带宽打满

TS="$(date +%Y%m%d_%H%M%S)"
RESULT_BASE="./bench_results/deepseek_v3.2_p800_vllm_${TS}"
SERVED_MODEL_NAME="base_model"
HOST="0.0.0.0"
PORT="38200"
MULTI_ROUND_DEFAULT=8
PERCENTILE_METRICS="ttft,tpot,itl,e2el"

# ===================== 核心测试函数 =====================
# 函数说明：封装vLLM基准测试逻辑，参数与原脚本完全对齐
# $1: case名称  $2: 输入token长度(isl)  $3: 输出token长度(osl)  $4: 并发数列表（空格分隔）
run_case () {
local case_name="$1"
local isl="$2"
local osl="$3"
local conc_list="$4"

# 每个case独立输出目录
local out_dir="${RESULT_BASE}/${case_name}"
mkdir -p "${out_dir}"

# 遍历所有并发数（对应原脚本的workers）
for c in ${conc_list}; do
# 计算num_prompts：并发数*乘数（>128时乘数=4，否则=8）
local concurrency_multiplier=${MULTI_ROUND_DEFAULT}
if [ ${c} -gt 128 ]; then
concurrency_multiplier=4
fi
local num_prompts=$((c * concurrency_multiplier))

echo "=== Case=${case_name} isl=${isl} osl=${osl} concurrency=${c} num_prompts=${num_prompts} ==="

# vLLM bench serve 核心命令（对齐原脚本所有参数逻辑）
vllm bench serve \
--backend vllm \
--model "${MODEL_PATH}" \
--served-model-name base_model \
--dataset-name random \
--random-input-len "${isl}" \
--random-output-len "${osl}" \
--num-prompts "${num_prompts}" \
--max-concurrency "${c}" \
--host "${HOST}" \
--port "${PORT}" \
--ignore-eos \
|& tee "${out_dir}/deepseek_v3.2__${case_name}_concurrency_${c}.log"
done

}

run_case "1005_635" 1005 635 "32 64 128 256"
run_case "200_1024" 200 1024 "32 64 128 256"
run_case "1024_1024" 1024 1024 "32 64 128 256"

#!/bin/bash
set -euo pipefail

MODEL_PATH="/worker/models/Qwen2.5-72B/"
TS="$(date +%Y%m%d_%H%M%S)"
RESULT_BASE="./bench_results/qwen2.5_72B_p800_vllm_${TS}"
SERVED_MODEL_NAME="base_model"
HOST="0.0.0.0"
PORT="8000"
MULTI_ROUND_DEFAULT=8
PERCENTILE_METRICS="ttft,tpot,itl,e2el"

# ===================== 核心测试函数 =====================
# 函数说明：封装vLLM基准测试逻辑，参数与原脚本完全对齐
# $1: case名称  $2: 输入token长度(isl)  $3: 输出token长度(osl)  $4: 并发数列表（空格分隔）
run_case () {
local case_name="$1"
local isl="$2"
local osl="$3"
local conc_list="$4"

# 每个case独立输出目录
local out_dir="${RESULT_BASE}/${case_name}"
mkdir -p "${out_dir}"

# 遍历所有并发数（对应原脚本的workers）
for c in ${conc_list}; do
# 计算num_prompts：并发数*乘数（>128时乘数=4，否则=8）
local concurrency_multiplier=${MULTI_ROUND_DEFAULT}
if [ ${c} -gt 128 ]; then
concurrency_multiplier=4
fi
local num_prompts=$((c * concurrency_multiplier))

echo "=== Case=${case_name} isl=${isl} osl=${osl} concurrency=${c} num_prompts=${num_prompts} ==="

# vLLM bench serve 核心命令（对齐原脚本所有参数逻辑）
vllm bench serve \
--backend vllm \
--model "${MODEL_PATH}" \
--tokenizer "${MODEL_PATH}" \
--dataset-name random \
--random-input-len "${isl}" \
--random-output-len "${osl}" \
--num-prompts "${num_prompts}" \
--max-concurrency "${c}" \
--host "${HOST}" \
--port "${PORT}" \
--ignore-eos \
--percentile-metrics "${PERCENTILE_METRICS}" \
--save-result \
--result-dir "${out_dir}" \
--result-filename "concurrency_${c}.json" \
|& tee "${out_dir}/qwen2.5_72B_${case_name}_concurrency_${c}.log"
done

}

#!/usr/bin/env bash
set -euo pipefail

# 1. 匹配你启动脚本里的 host 和 port
HOST="127.0.0.1"
PORT="9090"

# 2. 匹配启动脚本里的 --served-model-name 和 lora 别名
BASE_MODEL_NAME="Qwen2.5-VL-72B-Instruct"
LORA1_NAME="lora1"
LORA2_NAME="lora2"

# 你要问的一句话
QUESTION="总结一段关于商业的内容和小红帽大灰狼的故事，和42号混凝土的故事，需要包括国产DCU。共计100字"

call_model() {
local model_name="$1"
local question="$2"

# 使用 Python 处理 JSON 请求和响应
python3 - "$HOST" "$PORT" "$model_name" "$question" <<'PY'
import json, sys, subprocess

host, port, model, question = sys.argv[1:5]
url = f"http://{host}:{port}/v1/chat/completions"

# 构造 OpenAI 标准格式的 Payload
payload = {
"model": model,
"messages": [{"role": "user", "content": question}],
"temperature": 0.0,
"max_tokens": 512,
}

# 发送请求
p = subprocess.run(
["curl", "-s", url, "-H", "Content-Type: application/json",
"-d", json.dumps(payload, ensure_ascii=False)],
capture_output=True, text=True
)

out = p.stdout.strip()
try:
data = json.loads(out)
# 解析 OpenAI 响应格式
if "choices" in data:
print(data["choices"][0]["message"]["content"].strip())
else:
print(f"API ERROR: {data.get('message', 'Unknown error')}")
except Exception as e:
print(f"ERROR: Failed to parse JSON. Raw output: {out[:500]}")
sys.exit(1)
PY
}

echo "Question: $QUESTION"
echo "---------------------------------------------------"

echo "=== Testing Base Model: $BASE_MODEL_NAME ==="
call_model "$BASE_MODEL_NAME" "$QUESTION"
echo
echo "---------------------------------------------------"

echo "=== Testing LoRA 1: $LORA1_NAME ==="
call_model "$LORA1_NAME" "$QUESTION"
echo
echo "---------------------------------------------------"

echo "=== Testing LoRA 2: $LORA2_NAME ==="
call_model "$LORA2_NAME" "$QUESTION"
echo

#!/bin/bash
set -euo pipefail

export HF_HUB_OFFLINE=1
export TRANSFORMERS_OFFLINE=1
export HF_DATASETS_OFFLINE=1

# 创建空文件（如果不存在）
mkdir -p /tmp
touch /tmp/ShareGPT_V3_unfiltered_cleaned_split.json

# ===================== 配置参数 =====================
MODEL_NAME="/data1/models/GLM-5-w4a8/"
HOST="127.0.0.1"
PORT="30000"
TS="$(date +%Y%m%d_%H%M%S)"
RESULT_BASE="./bench_results/sglang_${MODEL_NAME}_${TS}"
MULTI_ROUND_DEFAULT=8

# ===================== 预热函数 =====================
warmup_service() {
echo "=== Warming up service (20 requests, concurrency 10) ==="
python3 -m sglang.bench_serving \
--backend sglang \
--model "${MODEL_NAME}" \
--host "${HOST}" \
--port "${PORT}" \
--dataset-name random \
--random-input-len 5000 \
--random-output-len 1800 \
--num-prompts 20 \
--max-concurrency 10 \
--request-rate inf \
--output-file /tmp/warmup.json \
> /dev/null 2>&1
echo "=== Warmup completed ==="
sleep 5
}

# ===================== 核心测试函数 =====================
run_case() {
local case_name="$1"
local isl="$2"
local osl="$3"
local conc_list="$4"

local out_dir="${RESULT_BASE}/${case_name}"
mkdir -p "${out_dir}"

for c in ${conc_list}; do
local concurrency_multiplier=${MULTI_ROUND_DEFAULT}
if [ ${c} -gt 128 ]; then
concurrency_multiplier=4
fi
local num_prompts=$((c * concurrency_multiplier))

echo "=== Case=${case_name} isl=${isl} osl=${osl} concurrency=${c} num_prompts=${num_prompts} ==="

# 添加固定输入长度，避免随机波动
python3 -m sglang.bench_serving \
--backend sglang \
--model "${MODEL_NAME}" \
--host "${HOST}" \
--port "${PORT}" \
--dataset-name random \
--random-input-len "${isl}" \
--random-output-len "${osl}" \
--random-range-ratio 0 \
--num-prompts "${num_prompts}" \
--max-concurrency "${c}" \
--request-rate inf \
--output-file "${out_dir}/concurrency_${c}.json" \
|& tee "${out_dir}/sglang_${case_name}_concurrency_${c}.log"

sleep 5
done
}

# ===================== 执行测试 =====================
mkdir -p "${RESULT_BASE}"
echo "Results will be saved to: ${RESULT_BASE}"

# 1. 先预热服务
warmup_service

# 2. 再跑正式测试
run_case "5k_1.8k" 5000 1800 "1 10 20 30 50 100"

echo "=== All tests completed ==="
echo "Results saved in: ${RESULT_BASE}"

#!/bin/bash
set -x
pkill -9 -f "sglang.launch_server|sglang serve" || true
sleep 2

MODEL_DIR=${model_dir:-/data1/models/MiniMax-M3-int4}
PORT=30000
DRAFT_MODEL_DIR=${draft_model_dir:-/data1/models/MiniMax-M3-EAGLE3}
ROOT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && cd ../../ && pwd)"
source "/workspace"/common/common.sh

# export XPU_ENABLE_PROFILER_TRACING=1
# export BKCL_FORCE_SYNC=1
# export SGLANG_TORCH_PROFILER_DIR=/workspace/xSGL/torch_profile

# export XSGL_REDIRCT_LOG=1
# export XPUAPI_DEBUG=0x1

########  上下文配置  ########
# 模型能力：MiniMax-M3 原生支持最长 1M（1048576）上下文（config.json text_config.max_position_embeddings=1048576）。
# 本机限制：单卡 96GB 显存无法承载完整 1M token 的 KV，原因如下——
#   w8a8 权重分片 ~51.3GB/卡；KV(fp16) ~44KB/token（60 层主注意力 K+V ~30KB + 57 层稀疏 index-K ~14KB）。
#   1M token KV = 1048576 × 44KB ≈ 46.3GB/卡，权重 51.3 + KV 46.3 ≈ 97.6GB > 96GB，
#   即便 mem_fraction_static 拉到 1.0（无运行时余量，必然 OOM）也放不下；
#   再扣掉激活/attention workspace/CUDA graph 捕获/通信 buffer 约 12GB 运行时，单卡 KV 实际上限约 ~740K token。
# 因此本脚本档位上限取 512K（单卡可容纳完整单请求 KV 的安全档）：
# CONTEXT_LEVEL=0    使用 512K profile（长上下文，低并发）；CONTEXT_LEVEL=1 使用 128K profile（均衡）；CONTEXT_LEVEL=2 使用 32K profile（高并发，高吞吐）。
# 显存预算：pool 容量 = mem_fraction_static*96 - 51.3(权重)；其余显存留给运行时（激活/attention/CUDA graph 捕获）。
#   长上下文运行时占用更大，故下调 mem_fraction_static 以保留运行时余量，避免捕获/前向 OOM。
#   各档 max_total_tokens 即 KV pool 大小（共享并发预算）；context_length 仅决定单请求最大长度。
# 仍可通过 context_length/max_total_tokens/max_running_requests/cuda_graph_max_bs/page_size/chunked_prefill_size/mem_fraction_static 等环境变量显式覆盖。

CONTEXT_LEVEL=${CONTEXT_LEVEL:-0}
PAGE_SIZE=${page_size:-128}

case "${CONTEXT_LEVEL}" in
0)
# 512K 长上下文，低并发：优先单请求长度。
# 单卡可容纳完整 512K 单请求 KV；pool=704K 允许 1 个满长 512K 请求 + ~192K 余量做中等并发。
# 512K 上下文在 mem=0.9 已实测可捕获，这里取 0.88 多留 ~2GB 运行时余量（长上下文 decode attention workspace 更大）。
CONTEXT_LENGTH=${context_length:-524288}
MAX_TOTAL_TOKENS=${max_total_tokens:-720896}
MAX_RUNNING_REQUESTS=${max_running_requests:-24}
CUDA_GRAPH_MAX_BS=${cuda_graph_max_bs:-24}
CHUNKED_PREFILL_SIZE=${chunked_prefill_size:-4096}
MEM_FRACTION_STATIC=${mem_fraction_static:-0.88}
PIECEWISE_CUDA_GRAPH_MAX_TOKENS=${piecewise_cuda_graph_max_tokens:-512}
PIECEWISE_CUDA_GRAPH_TOKENS=${piecewise_cuda_graph_tokens:-"4 8 16 32 64 128 192 256 384 512"}
;;
1)
# 128K 均衡：中等并发与吞吐。该档原最保守（空闲显存最多），顶到与 L0 相同的静态占用（mem 0.88/pool 704K）：
# L0 已在此 footprint 实测可启动，而 128K 前向/捕获占用比 512K 更小，故对本档严格更安全（实际空闲 ~14.2GB）。
# pool 512K→704K(+37%) 提并发预算；graph_bs 24→32、pcg 2048→4096 覆盖整个 prefill chunk，提升 decode/prefill 吞吐。
CONTEXT_LENGTH=${context_length:-262144} # 131072
MAX_TOTAL_TOKENS=${max_total_tokens:-720896}
MAX_RUNNING_REQUESTS=${max_running_requests:-64}
CUDA_GRAPH_MAX_BS=${cuda_graph_max_bs:-64}
CHUNKED_PREFILL_SIZE=${chunked_prefill_size:-4096}
MEM_FRACTION_STATIC=${mem_fraction_static:-0.88}
PIECEWISE_CUDA_GRAPH_MAX_TOKENS=${piecewise_cuda_graph_max_tokens:-4096}
PIECEWISE_CUDA_GRAPH_TOKENS=${piecewise_cuda_graph_tokens:-"4 8 16 32 64 128 192 256 384 512 768 1024 1536 2048 3072 4096"}
;;
2)
# 32K 短上下文，高并发/高吞吐：短上下文运行时占用小，故上调 mem 换取更大 KV pool 以提升并发与吞吐。
# pool=768K 在 mem=0.9 下 KV≈33.2GB，权重 51.3+KV 33.2=84.5GB < 静态预算 86.4GB；运行时余量 ~9.6GB，
# 32K 上下文前向/捕获占用远小于已验证可跑的 512K 档，故该余量充足。
# pcg 捕获到 4096（=chunked_prefill_size），让短请求 prefill 走 piecewise graph，吞吐更高。
CONTEXT_LENGTH=${context_length:-32768}
MAX_TOTAL_TOKENS=${max_total_tokens:-786432}
MAX_RUNNING_REQUESTS=${max_running_requests:-128}
CUDA_GRAPH_MAX_BS=${cuda_graph_max_bs:-128}
CHUNKED_PREFILL_SIZE=${chunked_prefill_size:-4096}
MEM_FRACTION_STATIC=${mem_fraction_static:-0.9}
PIECEWISE_CUDA_GRAPH_MAX_TOKENS=${piecewise_cuda_graph_max_tokens:-4096}
PIECEWISE_CUDA_GRAPH_TOKENS=${piecewise_cuda_graph_tokens:-"4 8 16 32 64 128 192 256 384 512 768 1024 1536 2048 3072 4096"}
;;
*)
echo "Invalid CONTEXT_LEVEL=${CONTEXT_LEVEL}. Valid values: 0(512K), 1(128K), 2(32K)."
exit 1
;;
esac
